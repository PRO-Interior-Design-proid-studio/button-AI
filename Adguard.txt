! Выбор провайдера :cxb по умолчанию
plusvibeapi.ru#%#(() => {
  const init = () => {
    const sel = document.querySelector('select[aria-label="Провайдер"]');
    if (sel && !sel.dataset.cxbSet) {
      sel.dataset.cxbSet = '1';
      sel.value = 'deepseek-v4.1-flash-1017:cxb';
      sel.dispatchEvent(new Event('change', { bubbles: true }));
    }
  };
  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', init);
  } else {
    init();
  }
})();
